ALTER TABLE fuel_rates ADD COLUMN is_active BOOLEAN DEFAULT true;

-- Set only the latest rate per fuel_type per city as active
WITH ranked_rates AS (
  SELECT 
    id,
    ROW_NUMBER() OVER (PARTITION BY fuel_type, city ORDER BY effective_date DESC) as rn
  FROM fuel_rates
)
UPDATE fuel_rates
SET is_active = CASE WHEN id IN (SELECT id FROM ranked_rates WHERE rn = 1) THEN true ELSE false END;

-- Helper function to check admin role
CREATE OR REPLACE FUNCTION is_admin()
RETURNS BOOLEAN AS $$
  SELECT EXISTS (
    SELECT 1
    FROM users
    WHERE id = auth.uid()
      AND role IN ('ADMIN', 'SUPERADMIN')
      AND is_active = TRUE
  );
$$ LANGUAGE sql STABLE;

-- =====================
-- TRIGGER: Prevent staff approval
-- =====================

CREATE OR REPLACE FUNCTION prevent_staff_approval()
RETURNS TRIGGER AS $$
BEGIN
  IF NEW.status IN ('APPROVED', 'REJECTED')
     AND NOT is_admin() THEN
    RAISE EXCEPTION 'Only admins can approve or reject cost sheets';
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_prevent_staff_approval
BEFORE UPDATE ON cost_sheets
FOR EACH ROW
EXECUTE FUNCTION prevent_staff_approval();
